cmake_minimum_required(VERSION 3.20)
project(bbport_graphics_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(BB_ROOT "../.." ABSOLUTE)
set(BB_SHADER_INCLUDE "${BB_ROOT}/out/windows/include" CACHE PATH "Generated host shader headers")
set(BB_VMA_INCLUDE "" CACHE PATH "Directory containing vk_mem_alloc.h")
option(FFX_VK_PORTABLE_BUILD_TESTS "" OFF)
option(FFX_VK_PORTABLE_BUILD_PROBE "" OFF)
option(FFX_VK_PORTABLE_BUILD_FSR3_HOST "" OFF)
option(FFX_VK_PORTABLE_BUILD_FSR3_VK_BACKEND "" OFF)
option(FFX_VK_PORTABLE_INSTALL "" OFF)
find_package(Vulkan REQUIRED)
add_subdirectory("${BB_ROOT}/gpu/third_party/fsr-vulkan" fsr-vulkan)
add_executable(fg-test ../test_frame_generation.cpp
    "${BB_ROOT}/gpu/shadps4/video_core/renderer_vulkan/vk_frame_generation.cpp"
    "${BB_ROOT}/gpu/shim/bbport_settings.cpp" "${BB_ROOT}/gpu/shim/bbport_background.cpp")
target_include_directories(fg-test PRIVATE "${BB_ROOT}/gpu/shim" "${BB_ROOT}/gpu/shadps4" "${BB_SHADER_INCLUDE}" "${BB_VMA_INCLUDE}")
target_compile_options(fg-test PRIVATE -UNDEBUG)
target_link_libraries(fg-test PRIVATE ffx-vulkan::fsr3-vk-framegeneration-3.1.6 ${CMAKE_DL_LIBS})
add_executable(xess-test ../test_xess.cpp "${BB_ROOT}/gpu/shadps4/video_core/renderer_vulkan/vk_xess.cpp" "${BB_ROOT}/gpu/shim/bbport_settings.cpp")
target_include_directories(xess-test PRIVATE "${BB_ROOT}/gpu/shim" "${BB_ROOT}/gpu/shadps4" "${Vulkan_INCLUDE_DIRS}")
target_compile_definitions(xess-test PRIVATE BB_XESS_TESTING)
target_compile_options(xess-test PRIVATE -UNDEBUG)
target_link_libraries(xess-test PRIVATE Vulkan::Vulkan)
